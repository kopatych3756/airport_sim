cmake_minimum_required(VERSION 3.16)
project(airport_runway_sim C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)

set(SOURCES src/main.c src/config.c src/io.c src/simulation.c src/sync.c)
add_executable(airport_sem ${SOURCES})
add_executable(airport_cond ${SOURCES})
target_compile_definitions(airport_cond PRIVATE USE_CONDVAR)
foreach(program airport_sem airport_cond)
    target_link_libraries(${program} PRIVATE Threads::Threads)
    target_compile_options(${program} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
