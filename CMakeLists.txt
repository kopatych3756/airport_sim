cmake_minimum_required(VERSION 3.16)
project(airport_runway_hw3 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
add_library(airport_core OBJECT src/config.c src/io.c src/model.c src/simulation.c src/ipc.c)
foreach (mode named unnamed shm fifo)
    add_executable(airport_${mode} src/main.c $<TARGET_OBJECTS:airport_core>)
    target_compile_definitions(airport_${mode} PRIVATE DEFAULT_MODE="${mode}")
    target_link_libraries(airport_${mode} PRIVATE pthread rt)
endforeach ()
add_executable(airport src/main.c $<TARGET_OBJECTS:airport_core>)
target_link_libraries(airport PRIVATE pthread rt)
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if (Python3_FOUND)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/run_tests.py)
    set_tests_properties(integration PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR} TIMEOUT 180)
endif ()
